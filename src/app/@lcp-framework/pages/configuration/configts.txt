import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { GridApiService } from '../../service/common/grid.service';
import { LocalStorageService } from '../../service/common/local-storage.service';
import { NgSelectModule } from '@ng-select/ng-select';

interface TabConfiguration {
  id: number;
  config_key: string;
  category_id: number;
  config_value: string;
  config_value_type: string;
  config_field_type: string;
}

interface Tab {
  id: number;
  name: string;
  configurations: TabConfiguration[];
}

@Component({
  selector: 'app-configuration',
  standalone: true,
  imports: [CommonModule, FormsModule, NgSelectModule],
  templateUrl: './configuration.component.html',
  styleUrls: ['./configuration.component.scss'],
})
export class ConfigurationComponent implements OnInit {
  userId: number | null = null;
  companyId: number | null = null;
  tabs: Tab[] = [];
  currentTab: string = '';
  fileInputs: { [key: string]: File | null } = {};

  gridpaginationdropdownList = ['5', '10', '15', '20', '25', '30', '40', '50', '60', '70', '80', '90', '100'];

  constructor(private gridApiService: GridApiService, public localStorageService: LocalStorageService) {}

  ngOnInit() {
    this.initializeUserData();
    this.loadAllItems(1);
  }

  initializeUserData() {
    const userData = this.localStorageService.getData('user_data');
    if (userData) {
      const parsedData = JSON.parse(userData);
      this.userId = parsedData.main?.id;
      this.companyId = parsedData.main?.company_id;
    }
  }

  applydropdownchanges(value: any) {
    return value.split(',');
  }

  loadAllItems(appCategoryTypeId: number) {
    if (!this.companyId) {
      console.error('Company ID is not available');
      return;
    }

    const payload = {
      company_id: this.companyId,
      print_query: true,
      primary_table: 'app_categories',
      start_index: 0,
      limit_range: 100,
      sort_columns: [['app_categories.id', 'asc']],
      search_all: [
        {
          column_name: 'app_categories.category_type_id',
          value: appCategoryTypeId,
          operator: '=',
        },
      ],
      select_columns: [
        ['app_categories.*'],
        [
          "CASE WHEN COUNT(app_configurations.id) = 0 THEN null ELSE COALESCE(Json_agg(DISTINCT jsonb_build_object('id', app_configurations.id,'config_key', app_configurations.config_key,'category_id', app_configurations.category_id,'config_value', app_configurations.config_value,'config_value_type', app_configurations.config_value_type,'config_field_type', app_configurations.config_field_type))) END",
          'configurations',
        ],
      ],
      includes: [
        {
          table_name: 'app_configurations',
          join_type: 'INNER',
          join_condition: `app_categories.id = app_configurations.category_id`,
        },
      ],
      group_by: ['app_categories.id'],
    };

    this.gridApiService.getAllList(payload).subscribe({
      next: (response: any) => {
        if (response.code === 200 && response.status) {
          console.log(response.data.records);
          if (response.data.records[1].configurations) {
            response.data.records[1].configurations.map(function (elem: any) {
              console.log(elem);
              if (elem.config_key == 'grid_pagination_dropdown') {
                elem.config_value = [10, 25, 50, 100];
              }
            });
          }
          this.tabs = response.data.records;
          if (this.tabs.length > 0) {
            this.currentTab = this.tabs[0].name;
          }
        }
      },
      error: (error) => {
        console.error('Error fetching entity types:', error);
      },
    });
  }

  onFileChange(event: Event, configKey: string) {
    const element = event.target as HTMLInputElement;
    const file = element.files ? element.files[0] : null;
    this.fileInputs[configKey] = file;
  }

  submitForm() {
    console.log('Form submitted');
    console.log('File inputs:', this.fileInputs);
    // Implement form submission logic here
  }

  switchTab(tabName: string) {
    this.currentTab = tabName;
  }
}
